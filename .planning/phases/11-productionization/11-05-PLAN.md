## 1. Objective

Deliver a secure frontend architecture with zero credential leaks into client bundles:
1. Centralized API Fetch Client: Implement `apiFetch` in `frontend/lib/api.ts` that automatically reads the `metaradar_csrf` cookie, attaches `X-CSRF-Token` on all mutating HTTP requests, includes credentials, and handles bootstrap via `GET /auth/csrf` when needed.
2. Demo Login without Password Leaks: Upgrade `DemoOperatorSelector` to call `POST /api/v1/auth/demo-login` with `{"role": "..."}`. Zero passwords placed in `NEXT_PUBLIC_*` or client-side code.
3. Review Queue Component: Build `frontend/components/signals/ReviewQueue.tsx` with role-aware queue filtering and fast decision controls.
4. Signal Status Lifecycle Visualization: Add real-time status chips and inline audit history to `SignalCard` and `SignalDetailWorkspace`.

## 2. Implementation Specifications

### A. [MODIFY] `frontend/lib/api.ts`

```typescript
function getCsrfToken(): string | null {
  if (typeof document === 'undefined') return null
  const match = document.cookie.match(new RegExp('(^| )metaradar_csrf=([^;]+)'))
  return match ? decodeURIComponent(match[2]) : null
}

export async function apiFetch<T>(endpoint: string, options: RequestInit = {}): Promise<T> {
  const url = `${API_BASE}${endpoint}`
  const headers = new Headers(options.headers || {})
  
  if (options.body && !headers.has('Content-Type')) {
    headers.set('Content-Type', 'application/json')
  }

  // Attach session-bound CSRF token on mutating requests
  const method = (options.method || 'GET').toUpperCase()
  if (['POST', 'PUT', 'PATCH', 'DELETE'].includes(method)) {
    const csrf = getCsrfToken()
    if (csrf) {
      headers.set('X-CSRF-Token', csrf)
    }
  }

  const response = await fetch(url, {
    ...options,
    headers,
    credentials: 'include', // Always send HttpOnly session & CSRF cookies
  })

  if (!response.ok) {
    let errorDetail = response.statusText
    try {
      const errJson = await response.json()
      errorDetail = errJson.detail || errorDetail
    } catch {
      // Non-JSON error body
    }
    throw new ApiError(response.status, errorDetail)
  }

  return response.json()
}

export const auth = {
  login: (email: string, password: string) =>
    apiFetch<UserMe>('/auth/login', {
      method: 'POST',
      body: JSON.stringify({ email, password }),
    }),

  demoLogin: (role: string) =>
    apiFetch<UserMe>('/auth/demo-login', {
      method: 'POST',
      body: JSON.stringify({ role }),
    }),

  logout: () =>
    apiFetch<{ status: string }>('/auth/logout', {
      method: 'POST',
    }),

  csrf: () => apiFetch<{ csrf_token: string }>('/auth/csrf'),

  me: () => apiFetch<UserMe>('/auth/me'),
}
```
