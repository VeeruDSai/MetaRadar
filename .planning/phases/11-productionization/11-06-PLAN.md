## 1. Objective

Harden application boundaries in strict alignment with OWASP security guidelines:
1. Database-Level Audit Immutability: Enforce append-only integrity via both PostgreSQL trigger and SQLAlchemy ORM event listeners. Any `UPDATE` or `DELETE` on `audit_log` raises a security exception.
2. Baseline Hardened CSP: Inject strict security headers with `default-src 'self'`, `object-src 'none'`, `base-uri 'none'`, `frame-ancestors 'none'`, `form-action 'self'`, `X-Content-Type-Options: nosniff`, and `X-Frame-Options: DENY`.
3. Pre-Auth Strict Exact-Origin Validation: Reject pre-auth requests (`POST /auth/login`, `POST /auth/demo-login`) when both `Origin` and `Referer` headers are absent, and validate exact scheme/host/port against `settings.cors_origins_list` (no prefix-based bypass).
4. Session-Bound HMAC CSRF: Validate `X-CSRF-Token` against the active session ID HMAC on all authenticated mutating requests.
5. Dedicated Security Suite: Comprehensive test suite in `tests/test_security.py` verifying headers, exact-origin validation, CSRF enforcement, trigger immutability, rate limiting, and security audit logs.

```python
class SecurityHeadersMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request: Request, call_next):
        response = await call_next(request)
        response.headers["X-Content-Type-Options"] = "nosniff"
        response.headers["X-Frame-Options"] = "DENY"
        response.headers["Referrer-Policy"] = "strict-origin-when-cross-origin"
        response.headers["X-XSS-Protection"] = "1; mode=block"
        response.headers["Content-Security-Policy"] = (
            "default-src 'self'; "
            "object-src 'none'; "
            "base-uri 'none'; "
            "frame-ancestors 'none'; "
            "form-action 'self'; "
            "script-src 'self' 'unsafe-inline'; "
            "style-src 'self' 'unsafe-inline' https://fonts.googleapis.com; "
            "font-src 'self' https://fonts.gstatic.com; "
            "img-src 'self' data: https:; "
            "connect-src 'self';"
        )
        return response
```

### B. [MODIFY] `backend/app/api/deps.py` — Strict Exact-Origin & Session-Bound CSRF Validation
```python
from urllib.parse import urlparse

def extract_origin(url: str) -> str:
    """Extracts exact scheme://netloc (including port if present) from a URL string."""
    parsed = urlparse(url)
    if not parsed.scheme or not parsed.netloc:
        return ""
    return f"{parsed.scheme}://{parsed.netloc}"

def validate_exact_origin(origin: str, allowed_origins: List[str]) -> bool:
    """Performs exact scheme/host/port comparison against allowed CORS origins."""
    normalized = origin.rstrip("/")
    allowed_set = {extract_origin(o).rstrip("/") if "://" in o else o.rstrip("/") for o in allowed_origins}
    return normalized in allowed_set

async def require_csrf(request: Request, current_user: User = Depends(get_current_user)) -> None:
    # Skip safe read operations
    if request.method in {"GET", "HEAD", "OPTIONS"}:
        return

    # Pre-auth login endpoints: Enforce mandatory Exact Origin / Referer validation
    if request.url.path in {"/api/v1/auth/login", "/api/v1/auth/demo-login"}:
        origin_header = request.headers.get("Origin")
        referer_header = request.headers.get("Referer")

        extracted_origin = None
        if origin_header:
            extracted_origin = extract_origin(origin_header)
        elif referer_header:
            extracted_origin = extract_origin(referer_header)
        else:
            # OWASP Guidance: Reject if neither Origin nor Referer is present
            raise HTTPException(
                status_code=403,
                detail="Forbidden: Missing Origin/Referer header on authentication request."
            )

        if not extracted_origin or not validate_exact_origin(extracted_origin, settings.cors_origins_list):
            raise HTTPException(
                status_code=403,
                detail="Forbidden: Untrusted request origin."
            )
        return

    # Authenticated mutating routes: Validate session-bound HMAC CSRF token
    cookie_token = request.cookies.get("metaradar_csrf", "")
    header_token = request.headers.get("X-CSRF-Token", "")
    if not cookie_token or not header_token or cookie_token != header_token:
        raise HTTPException(status_code=403, detail="Forbidden: CSRF token missing or mismatch.")

    token = request.cookies.get("metaradar_session", "")
    session_id_str = unsign_session_token(token, settings.SECRET_KEY, settings.SESSION_LIFETIME_SECONDS)
    if not verify_session_bound_csrf(header_token, session_id_str, settings.SECRET_KEY):
        raise HTTPException(status_code=403, detail="Forbidden: Invalid or forged session-bound CSRF token.")
```
