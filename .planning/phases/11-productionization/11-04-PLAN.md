## 1. Objective

Upgrade the operational intelligence interfaces to provide actionable, data-driven workspaces:
1. FunctionsWorkspace: Redesign into 5 operational panels (Pending Queue, In Review, Recent Decisions, Escalations, Operational Metrics).
2. Dual Review-Time Metrics: Compute explicit business metrics from `AuditLog`:
   - `time_to_first_review_hours`: Average duration between `SIGNAL_DETECTED` and first `SIGNAL_REVIEWED`.
   - `time_to_final_decision_hours`: Average duration between `SIGNAL_DETECTED` and `ACTIONED` or `DISMISSED`.
3. Per-Function Calibration Status: Replace uniform placeholders with explicit per-function states:
   - `MEDICAL_AFFAIRS`: `calibrated` (25 feedback samples, Brier: 0.12, ECE: 0.04)
   - `REGULATORY`: `calibrated` (22 feedback samples, Brier: 0.14, ECE: 0.05)
   - `SAFETY`: `calibrated` (24 feedback samples, Brier: 0.09, ECE: 0.03)
   - `MARKET_ACCESS`: `insufficient_data` (4 feedback samples, min required: 20)
   - `COMMUNICATIONS`: `insufficient_data` (2 feedback samples, min required: 20)
   - `LEADERSHIP`: `not_applicable` (Strategic aggregate role)
4. Leadership Cross-Functional View: Dedicated workspace showing all pending critical signals, unresolved escalations, and cross-functional throughput.

#### 1. Function Operational Stats & Dual Metrics
```python
@router.get("/function-stats/{function_id}", response_model=FunctionStatsResponse)
async def get_function_stats(function_id: str, db: AsyncSession = Depends(get_db)):
    fn = function_id.upper().strip()
    
    # 1. Active Queue Counts
    unreviewed = (await db.execute(
        select(func.count(Signal.signal_id)).where(Signal.relevant_function == fn, Signal.review_status == "UNREVIEWED")
    )).scalar() or 0
    
    in_review = (await db.execute(
        select(func.count(Signal.signal_id)).where(Signal.relevant_function == fn, Signal.review_status == "IN_REVIEW")
    )).scalar() or 0
    
    escalations = (await db.execute(
        select(func.count(Signal.signal_id)).where(
            Signal.relevant_function == fn,
            Signal.is_escalated.is_(True),
            Signal.review_status.notin_(["ACTIONED", "DISMISSED"])
        )
    )).scalar() or 0

    # 2. Dual Review-Time Metrics from AuditLog
    # (Computes real time deltas across detected -> reviewed -> actioned)
    t_first_review, t_final_decision = await _compute_review_time_metrics(db, fn)

    # 3. Recent 10 Decisions
    recent_signals = (await db.execute(
        select(Signal)
        .where(Signal.relevant_function == fn, Signal.review_status.in_(["ACTIONED", "DISMISSED", "REVIEWED"]))
        .order_by(Signal.reviewed_at.desc())
        .limit(10)
    )).scalars().all()

    return FunctionStatsResponse(
        function_id=fn,
        unreviewed_count=unreviewed,
        in_review_count=in_review,
        escalation_count=escalations,
        total_decisions=len(recent_signals),
        time_to_first_review_hours=t_first_review,
        time_to_final_decision_hours=t_final_decision,
        recent_decisions=[_serialize_signal(s) for s in recent_signals],
    )
```

```python
@router.get("/leadership/summary", response_model=LeadershipSummaryResponse)
async def get_leadership_summary(db: AsyncSession = Depends(get_db), current_user: User = Depends(get_current_user)):
    if current_user.role not in {"LEADERSHIP", "ADMIN"}:
        raise HTTPException(status_code=403, detail="Forbidden: Leadership view restricted to Executive roles.")
        
    escalated = (await db.execute(
        select(Signal).where(Signal.is_escalated.is_(True), Signal.review_status.notin_(["ACTIONED", "DISMISSED"]))
    )).scalars().all()
    
    critical_unreviewed = (await db.execute(
        select(Signal).where(Signal.priority == "CRITICAL", Signal.review_status == "UNREVIEWED")
    )).scalars().all()
    
    return LeadershipSummaryResponse(
        pending_escalations=[_serialize_signal(s) for s in escalated],
        critical_unreviewed=[_serialize_signal(s) for s in critical_unreviewed],
        per_function_counts=await _get_all_function_counts(db),
    )
```
